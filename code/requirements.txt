# Dependencies for the extraction, generation and verbalization steps.
# Install with: pip install -r requirements.txt
# Use a Python 3.10/3.11 environment; 3.14 may lack torch/transformers wheels.

# -- 1. activation extraction + 2. response generation --------------------
# (rq2_extract_activations.py / rq2_generate_responses.py)
torch                 # model forward pass on GPU; pick the CUDA build from pytorch.org
transformers>=4.50    # AutoTokenizer / AutoModelForCausalLM; Gemma-3 needs a recent version
accelerate            # required by device_map="auto"
pyarrow               # read/write the activation parquet files
numpy
pyyaml                # reads the layer index from nla_meta.yaml (with --nla-meta)
huggingface_hub       # download / offline load of models and checkpoints

# -- 3. NLA verbalization: AV verbalize + AR faithfulness ------------------
# (verbalize.py / score_roundtrip.py)
# The AV side needs a separate SGLang server hosting the AV checkpoint;
# the AR side runs in-process and needs no server.
sglang[all]>=0.5.6    # inference server hosting the AV checkpoint
httpx                 # verbalize.py talks to the SGLang server
orjson
safetensors           # reads the AV/AR checkpoint weights
pandas                # dedup in verbalize, convenience in score_roundtrip

# Step 3 also requires the Natural Language Autoencoders repository itself:
#   https://github.com/kitft/natural_language_autoencoders
