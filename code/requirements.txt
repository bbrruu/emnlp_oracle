# ── ① 抽 activation + ② 生成回答（rq2_extract_activations.py / rq2_generate_responses.py）──

sglang[all]>=0.5.6    # 掛 AV checkpoint 的推論 server（較重、環境相依）
httpx                 # verbalize 打 SGLang server
orjson
safetensors           # 讀 AV/AR checkpoint 權重
pandas                # verbalize finalize 去重 / score_roundtrip 便利
